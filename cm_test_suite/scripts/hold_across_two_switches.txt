# The control loop is held inside a read of sync hardware while two deactivations each run into
# their deadline. Afterwards both controllers can still be stopped, unloaded and loaded again.
switch strict -odom_pub hold
switch strict -jsb hold
switch strict -odom_pub
switch strict -jsb
unload odom_pub expect=accept
unload jsb expect=accept
load odom_pub expect=accept
load jsb expect=accept
